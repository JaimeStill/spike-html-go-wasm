// Command calc runs a tiny calculator language, one line at a time, over the script in input.txt.
//
// A line is an expression, such as (2 + 3) * 4, whose value it prints, or a binding,
// such as let rate = 0.07, which names a value for the lines after it and prints
// rate = 0.07. Numbers are decimal and may have a fraction; the operators are + - * /,
// with * and / binding tighter than + and -, a leading - to negate, and parentheses to
// group. A # starts a comment that runs to the end of the line.
//
// A line that fails prints input.txt:<line>: <reason> on stderr, and the run goes on with
// the next line. calc exits 1 if any line failed, and 0 otherwise.
//
// Each line goes through three steps, one file each: lexer.go splits it into tokens,
// parser.go builds a syntax tree from them, and eval.go works out its value.
package main

import (
	"fmt"
	"os"
	"strings"
)

// script is the file calc reads, from the working directory.
const script = "input.txt"

func main() {
	os.Exit(run())
}

// run runs the script, printing each line's result to stdout and each failure to stderr, and
// answers the exit code.
func run() int {
	src, err := os.ReadFile(script)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 1
	}
	vars := env{}
	code := 0
	for i, line := range strings.Split(string(src), "\n") {
		result, err := vars.exec(line)
		switch {
		case err != nil:
			fmt.Fprintf(os.Stderr, "%s:%d: %v\n", script, i+1, err)
			code = 1
		case result != "":
			fmt.Println(result)
		}
	}
	return code
}

package main

import (
	"fmt"
	"strconv"
	"strings"
	"unicode"
)

// A kind says what sort of token a token is.
type kind int

const (
	number   kind = iota // 4.25
	name                 // subtotal
	keyword              // let
	operator             // + - * / ( ) =
	end                  // the end of the line, after the last token
)

// A token is one word of a line: its kind, its text as written, and a number's value.
type token struct {
	kind  kind
	text  string
	value float64
}

// operators are the characters that stand alone as tokens.
const operators = "+-*/()="

// lex splits a line into its tokens, the last one always end. Spaces only separate tokens, and a
// # ends the line: everything after it is a comment.
func lex(line string) ([]token, error) {
	var tokens []token
	chars := []rune(line)
	for i := 0; i < len(chars); {
		c, start := chars[i], i
		switch {
		case c == '#':
			i = len(chars)
		case unicode.IsSpace(c):
			i++
		case unicode.IsDigit(c) || c == '.':
			for i < len(chars) && (unicode.IsDigit(chars[i]) || chars[i] == '.') {
				i++
			}
			text := string(chars[start:i])
			value, err := strconv.ParseFloat(text, 64)
			if err != nil {
				return nil, fmt.Errorf("bad number %q", text)
			}
			tokens = append(tokens, token{kind: number, text: text, value: value})
		case unicode.IsLetter(c) || c == '_':
			for i < len(chars) && (unicode.IsLetter(chars[i]) || unicode.IsDigit(chars[i]) || chars[i] == '_') {
				i++
			}
			t := token{kind: name, text: string(chars[start:i])}
			if t.text == "let" {
				t.kind = keyword
			}
			tokens = append(tokens, t)
		case strings.ContainsRune(operators, c):
			i++
			tokens = append(tokens, token{kind: operator, text: string(c)})
		default:
			return nil, fmt.Errorf("unexpected %q", string(c))
		}
	}
	return append(tokens, token{kind: end}), nil
}

package main

import (
	"errors"
	"fmt"
	"strings"
)

// An expr is a node of a line's syntax tree: a literal, a variable, a negation, or a binary
// operation, each holding the smaller expressions it is made of.
type expr any

type (
	// A literal is a number as written, such as 4.25.
	literal float64
	// A variable is a name to look up, such as subtotal.
	variable string
	// A negation is a leading minus, such as -discount.
	negation struct{ operand expr }
	// A binary is an operation on two expressions, such as price * 3.
	binary struct {
		op          string
		left, right expr
	}
)

// A statement is one line's syntax tree: an expression to print, bound to a name first when the
// line starts with let.
type statement struct {
	name  string // the name a let binds, or "" for a plain expression
	value expr
}

// parse builds the statement a line's tokens spell, or nil for a line with none, by this grammar,
// one method each, so * and / bind tighter than + and -:
//
//	statement = [ "let" name "=" ] sum
//	sum       = product { ( "+" | "-" ) product }
//	product   = factor { ( "*" | "/" ) factor }
//	factor    = "-" factor | number | name | "(" sum ")"
func parse(tokens []token) (*statement, error) {
	p := &parser{tokens: tokens}
	if p.peek().kind == end {
		return nil, nil
	}
	var s statement
	if p.peek().kind == keyword {
		p.next()
		t := p.next()
		if t.kind != name {
			return nil, unexpected(t)
		}
		s.name = t.text
		if t := p.next(); t.text != "=" {
			return nil, unexpected(t)
		}
	}
	value, err := p.sum()
	if err != nil {
		return nil, err
	}
	if t := p.next(); t.kind != end {
		return nil, unexpected(t)
	}
	s.value = value
	return &s, nil
}

// A parser reads a line's tokens from first to last.
type parser struct {
	tokens []token
	pos    int
}

// peek answers the next token without reading it.
func (p *parser) peek() token {
	return p.tokens[p.pos]
}

// next reads the next token; past the last, it answers end again.
func (p *parser) next() token {
	t := p.tokens[p.pos]
	if t.kind != end {
		p.pos++
	}
	return t
}

// sum parses products joined by + and -.
func (p *parser) sum() (expr, error) { return p.chain("+-", p.product) }

// product parses factors joined by * and /.
func (p *parser) product() (expr, error) { return p.chain("*/", p.factor) }

// chain parses operands joined by any of ops, grouping from the left, so 8 - 2 - 1 is 5.
func (p *parser) chain(ops string, operand func() (expr, error)) (expr, error) {
	left, err := operand()
	if err != nil {
		return nil, err
	}
	for p.peek().kind == operator && strings.Contains(ops, p.peek().text) {
		op := p.next().text
		right, err := operand()
		if err != nil {
			return nil, err
		}
		left = binary{op, left, right}
	}
	return left, nil
}

// factor parses a negation, a number, a name, or a sum in parentheses.
func (p *parser) factor() (expr, error) {
	t := p.next()
	switch {
	case t.text == "-":
		operand, err := p.factor()
		if err != nil {
			return nil, err
		}
		return negation{operand}, nil
	case t.kind == number:
		return literal(t.value), nil
	case t.kind == name:
		return variable(t.text), nil
	case t.text == "(":
		inner, err := p.sum()
		if err != nil {
			return nil, err
		}
		if t := p.next(); t.text != ")" {
			return nil, unexpected(t)
		}
		return inner, nil
	}
	return nil, unexpected(t)
}

// unexpected is the error for a token the grammar has no place for.
func unexpected(t token) error {
	if t.kind == end {
		return errors.New("unexpected end of line")
	}
	return fmt.Errorf("unexpected %q", t.text)
}

package main

import (
	"errors"
	"fmt"
	"strconv"
)

// An env holds the values that let has bound so far, by name, for the lines after.
type env map[string]float64

// exec runs one line: it lexes, parses, and evaluates it, binds a let's name, and answers what
// the line prints, or "" for a line with nothing to run. A line that fails binds nothing.
func (e env) exec(line string) (string, error) {
	tokens, err := lex(line)
	if err != nil {
		return "", err
	}
	s, err := parse(tokens)
	if err != nil || s == nil {
		return "", err
	}
	value, err := e.eval(s.value)
	if err != nil {
		return "", err
	}
	if s.name == "" {
		return format(value), nil
	}
	e[s.name] = value
	return s.name + " = " + format(value), nil
}

// errDivisionByZero is the error for dividing by zero, which would otherwise give an infinity.
var errDivisionByZero = errors.New("division by zero")

// eval works out an expression's value, its smaller expressions first.
func (e env) eval(x expr) (float64, error) {
	switch x := x.(type) {
	case literal:
		return float64(x), nil
	case variable:
		value, ok := e[string(x)]
		if !ok {
			return 0, fmt.Errorf("undefined: %s", x)
		}
		return value, nil
	case negation:
		value, err := e.eval(x.operand)
		if err != nil {
			return 0, err
		}
		return -value, nil
	case binary:
		left, err := e.eval(x.left)
		if err != nil {
			return 0, err
		}
		right, err := e.eval(x.right)
		if err != nil {
			return 0, err
		}
		switch x.op {
		case "+":
			return left + right, nil
		case "-":
			return left - right, nil
		case "*":
			return left * right, nil
		case "/":
			if right == 0 {
				return 0, errDivisionByZero
			}
			return left / right, nil
		}
	}
	panic(fmt.Sprintf("eval: no rule for %#v", x))
}

// format writes a value to 12 significant digits, enough to hide floating point's last-digit
// noise, so 0.1 + 0.2 prints 0.3. Negative zero, as 0 * -1 gives, prints 0: -0 == 0, so the
// comparison folds it into positive zero.
func format(value float64) string {
	if value == 0 {
		value = 0
	}
	return strconv.FormatFloat(value, 'g', 12, 64)
}

Your .go files build together as one main package, using only the standard library. There's no go.mod, no other packages of your own, no cgo or assembly, and nothing passed in as arguments or standard input.

Any other file you add sits next to your program, so os.ReadFile("input.txt") will find it. Each Run starts fresh from your files, so anything the program writes is gone by the next one.

Press Ctrl+Enter (Cmd+Enter on a Mac) to run, and Stop to cancel. The compiler stays loaded after a Stop, so the next Run is quick. Your edits live only in this tab: reload, and you're back to where you started.

The first Run downloads the compiler, the linker, and the standard-library packages your program imports; later Runs download only what a new import adds.
